# Test tree. Two CTest tiers so CI can gate correctness on every platform but keep the slow
# performance budget to a single Linux runner (docs/DEVELOPMENT_PLAN.md 2.3).

# Module mode, not CONFIG mode. The vcpkg gtest port builds GoogleTest and runs
# vcpkg_cmake_config_fixup(CONFIG_PATH lib/cmake/GTest), but across the three runner
# images CMake's config-package search did not resolve GTestConfig.cmake even with the
# installed tree on CMAKE_PREFIX_PATH. CMake's own FindGTest module searches the library
# and include directories directly, which is exactly the vcpkg layout, and it has
# defined GTest::gtest / GTest::gtest_main since CMake 3.20.
find_package(GTest REQUIRED)

add_library(openscrt_test_support INTERFACE)
target_link_libraries(openscrt_test_support INTERFACE GTest::gtest GTest::gtest_main
                                                 openscrt_warnings)

# openscrt_add_test(<name> LABEL <unit|perf> SOURCES <...> LIBS <...>)
function(openscrt_add_test name)
    cmake_parse_arguments(ARG "" "LABEL" "SOURCES;LIBS" ${ARGN})
    add_executable(${name} ${ARG_SOURCES})
    target_include_directories(${name} PRIVATE ${PROJECT_SOURCE_DIR}/src
                                                 ${PROJECT_SOURCE_DIR}/tests)
    target_link_libraries(${name} PRIVATE openscrt_test_support ${ARG_LIBS})
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES LABELS "${ARG_LABEL}" TIMEOUT 600)
endfunction()

# P1-A2: the VT state machine against a baseline, per case.
#
# The baselines in tests/baselines/vt are generated by tmux, an independent implementation
# of the same escape sequences. A baseline produced by our own emulator would only prove
# that we did not change; one produced by tmux is evidence about the sequences.
#
# This target brings its own main() so it can regenerate the baselines before the
# comparison runs (OPENCRT_VT_WRITE_BASELINES=<dir>), which is why it does not use
# openscrt_add_test: that helper links gtest_main, and two mains would not link.
add_executable(unit_vt_baseline unit/VtBaselineTest.cpp support/VtReference.cpp)
target_include_directories(unit_vt_baseline PRIVATE ${PROJECT_SOURCE_DIR}/src
                                                    ${PROJECT_SOURCE_DIR}/tests)
target_link_libraries(unit_vt_baseline PRIVATE GTest::gtest openscrt_terminal openscrt_protocols
                                                 openscrt_warnings)
target_compile_definitions(unit_vt_baseline PRIVATE
    OPENCRT_SOURCE_DIR="${PROJECT_SOURCE_DIR}")
add_test(NAME unit_vt_baseline COMMAND unit_vt_baseline)
set_tests_properties(unit_vt_baseline PROPERTIES LABELS "unit" TIMEOUT 600)

# P1-A4: the SPSC ring buffer under ThreadSanitizer.
#
# The sanitizer has to instrument the code under test, and the ring buffer is header-only,
# so the concurrency test gets its own target instead of a global flag: instrumenting the
# whole tree would slow every other test down without telling us anything the ring
# buffer's own accesses do not. GoogleTest stays uninstrumented on purpose -- it does not
# touch the ring.
#
# The target is built in its own build tree (preset linux-tsan) because it must NOT be built
# together with --coverage: gcov keeps its counters in plain globals, two threads increment
# the same one, and ThreadSanitizer reports that as a data race. It is a race in the coverage
# counters, not in the ring buffer, and the only way to keep the two apart is to not build
# them together.
option(OPENCRT_ENABLE_TSAN "Build the SPSC concurrency test with ThreadSanitizer" OFF)
if(OPENCRT_ENABLE_TSAN AND NOT MSVC)
    add_executable(unit_spsc_ring_buffer_tsan unit/SpscRingBufferTest.cpp)
    target_include_directories(unit_spsc_ring_buffer_tsan PRIVATE ${PROJECT_SOURCE_DIR}/src
                                                        ${PROJECT_SOURCE_DIR}/tests)
    target_link_libraries(unit_spsc_ring_buffer_tsan PRIVATE openscrt_test_support)
    target_compile_options(unit_spsc_ring_buffer_tsan PRIVATE -fsanitize=thread -g)
    target_link_options(unit_spsc_ring_buffer_tsan PRIVATE -fsanitize=thread)
    add_test(NAME unit_spsc_ring_buffer_tsan COMMAND unit_spsc_ring_buffer_tsan)
    set_tests_properties(unit_spsc_ring_buffer_tsan PROPERTIES LABELS "tsan" TIMEOUT 900)
endif()
